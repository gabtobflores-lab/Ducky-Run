# Proposed aurora shader (v5), not installed in the game until approved. Drop-in replacement for the aurora entry in mirror.html.
    aurora: `/* an original aurora blending Apple's Messages aurora with a real one: broad, soft sheets of light that sweep down the sky in a slow S, folding
   as they turn. Each sheet is a wide band around a smooth curve; where it turns we look along it, so it glows brighter and its lit edge sharpens.
   Fine vertical rays run through every sheet, as in real curtains. Colour moves across each sheet: a bright cyan-teal lit edge, blue in the body,
   lavender and violet in the folds behind. Deep navy all around; nothing has a hard line */
float n1(float x,float s){return vn(vec2(x,s))*2.-1.;}
float SLP,UU;
/* one sheet: x0 its centre, A how far the S swings, fr how tight, ph its phase, w its half width; returns its light; UU is the position across it */
float sheet(vec2 p,float x0,float A,float fr,float ph,float w,float sd,float y0,float y1){
  float y=p.y,arg=y*fr+ph+.35*n1(y*1.1+t*.02,sd);
  float cx=x0+A*sin(arg)+.018*n1(y*2.3-t*.03,sd+3.);
  float sl=A*fr*cos(arg);SLP=abs(sl);                                                        /* how steeply the sheet runs sideways here */
  float ww=w*(.8+.35*vn(vec2(y*1.6-t*.018,sd+5.)));
  float u=(p.x-cx)/ww/sqrt(1.+sl*sl);UU=u;                                                   /* measured across the sheet, so it keeps its width as it turns */
  float edge=exp(-pow((u-.45)/.5,2.));                                                      /* the lit edge */
  float body=exp(-u*u*.7)*.6+exp(-u*u*.18)*.12;                                             /* the soft body and a wide glow round it */
  float fold=1.+.9*smoothstep(.6,2.4,SLP);                                                  /* where it turns we look along it */
  float rays=.72+.28*smoothstep(.2,.9,vn(vec2(p.x*55.+sd*13.,y*1.1-t*.06)))*(.9+.1*vn(vec2(p.x*150.+sd,y*2.5)));  /* fine vertical rays */
  float env=smoothstep(y0,y0+.12,y)*(1.-smoothstep(y1-.14,y1,y));
  return (edge*fold*.75+body)*rays*env;}
void main(){vec2 uv=UV();float asp=res.x/res.y;vec2 p=vec2(uv.x*asp,uv.y);vec2 pp=vec2(p.x/(asp/.46),p.y);
  vec3 a=vec3(0.);
  vec3 CYAN=vec3(.35,.95,.95),TEAL=vec3(.25,.9,.72),BLUE=vec3(.1,.38,1.),VIO=vec3(.48,.32,1.),LAV=vec3(.72,.62,1.);
  /* the far sheet: violet and lavender, swinging the other way behind */
  float f=sheet(pp,.27,.085,5.2,2.6,.075,7.,.02,.82);vec3 fc=mix(VIO,LAV,smoothstep(-.2,.8,UU));a+=fc*f*.38;
  /* the main sheet: a great S from the top down to just above the balls */
  float m=sheet(pp,.22,.1,6.1,.4+t*.05,.085,3.,-.05,.74);
  vec3 mc=mix(VIO,BLUE,smoothstep(-1.2,-.2,UU));mc=mix(mc,CYAN,smoothstep(.1,.6,UU));mc=mix(mc,TEAL,smoothstep(.3,.9,UU)*(.4+.45*smoothstep(.3,.7,pp.y)));
  mc=mix(mc,vec3(.88,1.,1.),smoothstep(1.8,2.6,m)*.5);a+=mc*m*1.05;
  /* a smaller sheet high on the left, teal fading to blue */
  float s=sheet(pp,.08,.05,7.,1.2,.055,11.,-.05,.42);a+=mix(BLUE,TEAL,smoothstep(-.3,.7,UU))*s*.45;
  /* a low, faint fringe of rays hanging near the bottom, kept dim round the balls */
  { float fr=exp(-pow((pp.y-.92)/.07,2.))*(.7+.3*vn(vec2(pp.x*30.,pp.y*1.2-t*.05)));a+=mix(TEAL,BLUE,.55)*fr*.1; }
  /* the coloured air: blue haze round the light, violet low on the right, deep navy in the corners */
  vec3 bg=mix(vec3(.006,.012,.04),vec3(.02,.022,.08),uv.y);
  bg+=vec3(.03,.16,.42)*exp(-dot(pp-vec2(.22,.45),pp-vec2(.22,.45))/.05)*.28+vec3(.2,.12,.45)*exp(-dot(pp-vec2(.36,.9),pp-vec2(.36,.9))/.06)*.5;
  vec3 col=1.-exp(-(bg+a*1.5)*1.2);col=pow(col,vec3(1.12)); /* a little more contrast: deep gaps between the sheets */
  float lum=dot(col,vec3(.3,.5,.2));col+=starfield(p,.25)*(1.-smoothstep(.05,.2,lum));
  gl_FragColor=vec4(col+dith(),1.);}
